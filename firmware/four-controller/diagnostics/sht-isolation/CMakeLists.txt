cmake_minimum_required(VERSION 3.22)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(ctrl02_sht_isolation C ASM)
set(DIAG_BUILD_ID "development" CACHE STRING "Diagnostic source fingerprint")
get_filename_component(FW "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(BASE "${FW}/../stm32f103rct6" ABSOLUTE)
set(HAL "${BASE}/Drivers/STM32F1xx_HAL_Driver")
set(DIAG_I2C_HAL_DIR "${CMAKE_CURRENT_BINARY_DIR}/vendor" CACHE PATH "Pinned ST HAL v1.1.10 I2C dependency (see README)")
set(SOURCES main.c production_soft_i2c.c "${BASE}/Core/Src/system_stm32f1xx.c" "${BASE}/startup_stm32f103xe.s")
list(APPEND SOURCES "${DIAG_I2C_HAL_DIR}/Src/stm32f1xx_hal_i2c.c")
foreach(PART hal hal_cortex hal_gpio hal_gpio_ex hal_rcc hal_rcc_ex hal_pwr hal_flash hal_flash_ex hal_uart)
  list(APPEND SOURCES "${HAL}/Src/stm32f1xx_${PART}.c")
endforeach()
add_executable(ctrl02_sht_isolation.elf ${SOURCES})
target_include_directories(ctrl02_sht_isolation.elf PRIVATE "${DIAG_I2C_HAL_DIR}/Inc" "${FW}/Core/Inc" "${BASE}/Core/Inc" "${HAL}/Inc" "${HAL}/Inc/Legacy" "${BASE}/Drivers/CMSIS/Include" "${BASE}/Drivers/CMSIS/Device/ST/STM32F1xx/Include")
target_compile_definitions(ctrl02_sht_isolation.elf PRIVATE USE_HAL_DRIVER STM32F103xE HAL_I2C_MODULE_ENABLED BOARD_ID=2 UT_SHT_DIAGNOSTIC=1 DIAG_BUILD_ID="${DIAG_BUILD_ID}")
target_compile_options(ctrl02_sht_isolation.elf PRIVATE -mcpu=cortex-m3 -mthumb -Os -g3 -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
target_link_options(ctrl02_sht_isolation.elf PRIVATE -mcpu=cortex-m3 -mthumb "-T${BASE}/STM32F103xx_FLASH.ld" --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--print-memory-usage "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/ctrl02_sht_isolation.map")
add_custom_command(TARGET ctrl02_sht_isolation.elf POST_BUILD COMMAND arm-none-eabi-objcopy -O binary "$<TARGET_FILE:ctrl02_sht_isolation.elf>" "${CMAKE_CURRENT_BINARY_DIR}/ctrl02_sht_isolation.bin" COMMAND arm-none-eabi-size "$<TARGET_FILE:ctrl02_sht_isolation.elf>")
